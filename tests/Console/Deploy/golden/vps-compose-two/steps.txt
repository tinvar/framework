printf 'TINVAR_APP_KEY=%s\nTINVAR_PROBE_TOKEN=%s\nTINVAR_TRUSTED_PROXY_SECRET=%s\n' "$(cat '/home/you/.local/state/tinvar/my-app-2cab79174cf3.vps-compose.key')" "$(cat '/home/you/.local/state/tinvar/my-app-2cab79174cf3.vps-compose.probe-token')" "$(cat '/home/you/.local/state/tinvar/my-app-2cab79174cf3.vps-compose.proxy-secret')" | ssh SERVER 'sudo sh -c "umask 077 && mkdir -p /etc/my-app && cat > /etc/my-app/app.env"'
printf 'TINVAR_TRUSTED_PROXY_SECRET=%s\n' "$(cat '/home/you/.local/state/tinvar/my-app-2cab79174cf3.vps-compose.proxy-secret')" | ssh SERVER 'sudo sh -c "umask 077 && mkdir -p /etc/my-app && cat > /etc/my-app/proxy.env"'
